#include<bits/stdc++.h>
using namespace std;

int main(){
    
    int n;
    cin>>n;
    
    vector<int> arr(n);
    for(int i=0; i<n; i++)
    cin>>arr[i];
    
    long long ans=0;
    int last=arr[0];
    
    for(int i=1; i<n; i++){
        if(arr[i] < last){
            ans += (last-arr[i]);
        }
        else{
            last = arr[i];
        }
    }
    
    cout<<ans;
    
    return 0;
}
